import "fmt"

type Array struct {
        data     []string
        size     int
        capacity int
}

func NewArray(cap int) *Array {
        return &Array{
                data:     make([]string, cap),
                size:     0,
                capacity: cap,
        }
}

func (a *Array) resize() {
        newCapacity := 4
        if a.capacity > 0 {
                newCapacity = a.capacity * 2
        }
        newData := make([]string, newCapacity)
        copy(newData, a.data[:a.size])
        a.data = newData
        a.capacity = newCapacity
}

func (a *Array) AddEnd(text string) {
        if a.size == a.capacity {
                a.resize()
        }
        a.data[a.size] = text
        a.size++
}

func (a *Array) AddIndex(text string, index int) {
        if index < 0 || index > a.size {
                return
        }
        if a.size == a.capacity {
                a.resize()
        }
        for i := a.size; i > index; i-- {
                a.data[i] = a.data[i-1]
        }
        a.data[index] = text
        a.size++
}

func (a *Array) Get(index int) string {
        if index < 0 || index >= a.size {
                return ""
        }
        return a.data[index]
}

func (a *Array) RemoveIndex(index int) {
        if index < 0 || index >= a.size {
                return
        }
        for i := index; i < a.size-1; i++ {
                a.data[i] = a.data[i+1]
        }
        a.size--
}

func (a *Array) Set(index int, text string) {
        if index < 0 || index >= a.size {
                return
        }
        a.data[index] = text
}

func (a *Array) Length() int {
        return a.size
}

func (a *Array) Print() {
        for i := 0; i < a.size; i++ {
                fmt.Printf("%s ", a.data[i])
        }
        fmt.Println()
}

type CuckooEntry[T comparable] struct {
        key      T
        occupied bool
}

type CuckooHash[T comparable] struct {
        table1   []CuckooEntry[T]
        table2   []CuckooEntry[T]
        capacity int
        size     int
}

func NewCuckooHash[T comparable](cap int) *CuckooHash[T] {
        return &CuckooHash[T]{
                table1:   make([]CuckooEntry[T], cap),
                table2:   make([]CuckooEntry[T], cap),
                capacity: cap,
                size:     0,
        }
}

func (c *CuckooHash[T]) hashHelper(key T, salt string, mod int) int {
        hasher := fnv.New32a()
        hasher.Write([]byte(fmt.Sprintf("%v%s", key, salt)))
        return int(hasher.Sum32()) % mod
}

func (c *CuckooHash[T]) hash1(key T) int { return c.hashHelper(key, "", c.capacity) }
func (c *CuckooHash[T]) hash2(key T) int { return c.hashHelper(key, "salt", c.capacity) }

func (c *CuckooHash[T]) rehash() {
        old1 := c.table1
        old2 := c.table2
        oldCap := c.capacity

        c.capacity *= 2
        c.table1 = make([]CuckooEntry[T], c.capacity)
        c.table2 = make([]CuckooEntry[T], c.capacity)
        c.size = 0

        for i := 0; i < oldCap; i++ {
                if old1[i].occupied {
                        c.Add(old1[i].key)
                }
                if old2[i].occupied {
                        c.Add(old2[i].key)
                }
        }
}

func (c *CuckooHash[T]) Add(key T) {
        if c.Contains(key) {
                return
        }

        currentKey := key
        for i := 0; i < 100; i++ {
                pos1 := c.hash1(currentKey)
                if !c.table1[pos1].occupied {
                        c.table1[pos1] = CuckooEntry[T]{key: currentKey, occupied: true}
                        c.size++
                        return
                }
                // Выселяем
                currentKey, c.table1[pos1].key = c.table1[pos1].key, currentKey

                pos2 := c.hash2(currentKey)
                if !c.table2[pos2].occupied {
                        c.table2[pos2] = CuckooEntry[T]{key: currentKey, occupied: true}
                        c.size++
                        return
                }
                currentKey, c.table2[pos2].key = c.table2[pos2].key, currentKey
        }
        c.rehash()
        c.Add(currentKey)
}

func (c *CuckooHash[T]) Contains(key T) bool {
        pos1 := c.hash1(key)
        if c.table1[pos1].occupied && c.table1[pos1].key == key {
                return true
        }
        pos2 := c.hash2(key)
        if c.table2[pos2].occupied && c.table2[pos2].key == key {
                return true
        }
        return false
}

type hashNode[T comparable] struct {
        data T
        next *hashNode[T]
}

type HashSet[T comparable] struct {
        table    []*hashNode[T]
        capacity int
        size     int
}

func NewHashSet[T comparable](cap int) *HashSet[T] {
        return &HashSet[T]{
                table:    make([]*hashNode[T], cap),
                capacity: cap,
                size:     0,
        }
}

func (h *HashSet[T]) hash(key T) int {
        // Упрощенная хеш-функция через fmt.Sprintf для универсальности
        hasher := fnv.New32a()
        hasher.Write([]byte(fmt.Sprintf("%v", key)))
        return int(hasher.Sum32()) % h.capacity
}

func (h *HashSet[T]) Add(value T) {
        idx := h.hash(value)
        cur := h.table[idx]
        for cur != nil {
                if cur.data == value {
                        return // Уже есть
                }
                cur = cur.next
        }
        // Добавляем в начало цепочки
        h.table[idx] = &hashNode[T]{data: value, next: h.table[idx]}
        h.size++
}

func (h *HashSet[T]) Remove(value T) bool {
        idx := h.hash(value)
        cur := h.table[idx]
        var prev *hashNode[T]
        for cur != nil {
                if cur.data == value {
                        if prev != nil {
                                prev.next = cur.next
                        } else {
                                h.table[idx] = cur.next
                        }
                        h.size--
                        return true
                }
                prev = cur
                cur = cur.next
        }
        return false
}

func (h *HashSet[T]) Contains(value T) bool {
        idx := h.hash(value)
        cur := h.table[idx]
        for cur != nil {
                if cur.data == value {
                        return true
                }
                cur = cur.next
        }
        return false
}

func (h *HashSet[T]) GetSize() int {
        return h.size
}

func (h *HashSet[T]) GetAll() []T {
        result := make([]T, 0, h.size)
        for i := 0; i < h.capacity; i++ {
                cur := h.table[i]
                for cur != nil {
                        result = append(result, cur.data)
                        cur = cur.next
                }
        }
        return result
}

func (h *HashSet[T]) Clear() {
        for i := 0; i < h.capacity; i++ {
                h.table[i] = nil
        }
        h.size = 0
}

type QueueNode struct {
        text string
        next *QueueNode
}

type Queue struct {
        head *QueueNode
        tail *QueueNode
}

func NewQueue() *Queue {
        return &Queue{head: nil, tail: nil}
}

func (q *Queue) Push(value string) {
        newNode := &QueueNode{text: value, next: nil}
        if q.tail == nil {
                q.head = newNode
                q.tail = newNode
                return
        }
        q.tail.next = newNode
        q.tail = newNode
}

func (q *Queue) Pop(outValue *string) bool {
        if q.IsEmpty() {
                return false
        }
        *outValue = q.head.text
        q.head = q.head.next
        if q.head == nil {
                q.tail = nil
        }
        return true
}

func (q *Queue) IsEmpty() bool {
        return q.head == nil
}

func (q *Queue) Print() {
        if q.IsEmpty() {
                return
        }
        current := q.head
        for current != nil {
                fmt.Printf("%s", current.text)
                if current.next != nil {
                        fmt.Print(" -> ")
                }
                current = current.next
        }
        fmt.Println()
}

func (q *Queue) Clear() {
        q.head = nil
        q.tail = nil
}

type RBNode struct {
        Data   int
        IsRed  bool
        Left   *RBNode
        Right  *RBNode
        Parent *RBNode
}

func NewRBNode(val int, red bool, parent *RBNode) *RBNode {
        return &RBNode{Data: val, IsRed: red, Parent: parent}
}

type RedBlackTree struct {
        root *RBNode
}

func NewRedBlackTree() *RedBlackTree {
        return &RedBlackTree{root: nil}
}

func (t *RedBlackTree) rotateLeft(x *RBNode) {
        y := x.Right
        x.Right = y.Left
        if y.Left != nil {
                y.Left.Parent = x
        }
        y.Parent = x.Parent
        if x.Parent == nil {
                t.root = y
        } else if x == x.Parent.Left {
                x.Parent.Left = y
        } else {
                x.Parent.Right = y
        }
        y.Left = x
        x.Parent = y
}

func (t *RedBlackTree) rotateRight(x *RBNode) {
        y := x.Left
        x.Left = y.Right
        if y.Right != nil {
                y.Right.Parent = x
        }
        y.Parent = x.Parent
        if x.Parent == nil {
                t.root = y
        } else if x == x.Parent.Right {
                x.Parent.Right = y
        } else {
                x.Parent.Left = y
        }
        y.Right = x
        x.Parent = y
}

func (t *RedBlackTree) Insert(value int) {
        z := NewRBNode(value, true, nil)
        var y *RBNode
        x := t.root

        for x != nil {
                y = x
                if z.Data < x.Data {
                        x = x.Left
                } else if z.Data > x.Data {
                        x = x.Right
                } else {
                        return // Дубликат
                }
        }

        z.Parent = y
        if y == nil {
                t.root = z
        } else if z.Data < y.Data {
                y.Left = z
        } else {
                y.Right = z
        }

        t.fixInsert(z)
}

func (t *RedBlackTree) fixInsert(z *RBNode) {
        for z.Parent != nil && z.Parent.IsRed {
                if z.Parent == z.Parent.Parent.Left {
                        y := z.Parent.Parent.Right
                        if y != nil && y.IsRed {
                                z.Parent.IsRed = false
                                y.IsRed = false
                                z.Parent.Parent.IsRed = true
                                z = z.Parent.Parent
                        } else {
                                if z == z.Parent.Right {
                                        z = z.Parent
                                        t.rotateLeft(z)
                                }
                                z.Parent.IsRed = false
                                z.Parent.Parent.IsRed = true
                                t.rotateRight(z.Parent.Parent)
                        }
                } else {
                        y := z.Parent.Parent.Left
                        if y != nil && y.IsRed {
                                z.Parent.IsRed = false
                                y.IsRed = false
                                z.Parent.Parent.IsRed = true
                                z = z.Parent.Parent
                        } else {
                                if z == z.Parent.Left {
                                        z = z.Parent
                                        t.rotateRight(z)
                                }
                                z.Parent.IsRed = false
                                z.Parent.Parent.IsRed = true
                                t.rotateLeft(z.Parent.Parent)
                        }
                }
        }
        t.root.IsRed = false
}

func (t *RedBlackTree) Search(target int) bool {
        cur := t.root
        for cur != nil {
                if target == cur.Data {
                        return true
                } else if target < cur.Data {
                        cur = cur.Left
                } else {
                        cur = cur.Right
                }
        }
        return false
}

func (t *RedBlackTree) printInOrder(node *RBNode) {
        if node == nil {
                return
        }
        t.printInOrder(node.Left)
        fmt.Printf("%d ", node.Data)
        t.printInOrder(node.Right)
}

func (t *RedBlackTree) Print() {
        t.printInOrder(t.root)
        fmt.Println()
}

func (t *RedBlackTree) Clear() {
        t.root = nil
}
